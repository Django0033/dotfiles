** Mappings
*** Visual Line Navigation (k/j)
#+begin_src emacs-lisp
  (defun evil-move-visual-down (arg)
    (interactive "P")
    (if (or (null arg) (consp arg))
        (evil-next-visual-line 1)
      (evil-next-line (prefix-numeric-value arg))))

  (defun evil-move-visual-up (arg)
    (interactive "P")
    (if (or (null arg) (consp arg))
        (evil-previous-visual-line 1)
      (evil-previous-line (prefix-numeric-value arg))))

  (eval-after-load 'evil
    '(progn
  (define-key evil-normal-state-map "k" 'evil-move-visual-up)
  (define-key evil-normal-state-map "j" 'evil-move-visual-down)))
#+end_src

*** Move lines (Alt+j/k)
#+begin_src emacs-lisp
  (defun move-line-down ()
    (interactive)
    (let ((col (current-column)))
      (forward-line 1)
      (transpose-lines 1)
      (forward-line -1)
      (move-to-column col)))

  (defun move-line-up ()
    (interactive)
    (let ((col (current-column)))
      (transpose-lines 1)
      (forward-line -2)
      (move-to-column col)))

  (eval-after-load 'evil
    '(progn
       (define-key evil-normal-state-map (kbd "M-j") 'move-line-down)
       (define-key evil-normal-state-map (kbd "M-k") 'move-line-up)
       (define-key evil-visual-state-map (kbd "M-j") 'move-line-down)
       (define-key evil-visual-state-map (kbd "M-k") 'move-line-up)))
#+end_src
